## 1. 测试概述

### 1.1 测试目标

通过全面的性能基准测试，实现以下目标：

- **建立性能基线**: 确定系统在不同负载下的性能表现
- **识别性能瓶颈**: 发现影响系统性能的关键因素
- **验证优化效果**: 评估性能优化措施的有效性
- **容量规划**: 为系统扩容提供数据支持

### 1.2 测试原则

| 原则 | 说明 | 实施要求 |
|-----|-----|---------|
| 真实性 | 模拟真实业务场景 | 使用生产数据脱敏副本 |
| 可重复性 | 测试结果可复现 | 固定测试环境和参数 |
| 可比性 | 不同版本可对比 | 保持测试条件一致 |
| 全面性 | 覆盖多种场景 | 包含读写、并发、压力测试 |

### 1.3 测试范围

| 测试类型 | 测试内容 | 优先级 |
|---------|---------|-------|
| 基础性能测试 | 单表查询、索引查询 | ⭐⭐⭐⭐⭐ |
| 并发性能测试 | 多用户并发访问 | ⭐⭐⭐⭐⭐ |
| 压力测试 | 极限负载测试 | ⭐⭐⭐⭐ |
| 稳定性测试 | 长时间运行测试 | ⭐⭐⭐ |
| 恢复测试 | 故障恢复性能 | ⭐⭐⭐ |

### 3.1 基础性能测试

#### 3.1.1 单表查询测试

**测试目的**: 评估单表查询的基本性能

**测试SQL**:

```sql
-- 1. 主键查询
EXPLAIN (ANALYZE, BUFFERS, TIMING)
SELECT * FROM core.users WHERE id = '00000000-0000-0000-0000-000000000001';

-- 2. 唯一索引查询
EXPLAIN (ANALYZE, BUFFERS, TIMING)
SELECT * FROM core.users WHERE email = 'user@example.com';

-- 3. 范围查询
EXPLAIN (ANALYZE, BUFFERS, TIMING)
SELECT * FROM core.conversations 
WHERE created_at >= '2025-01-01' AND created_at < '2025-02-01'
ORDER BY created_at DESC LIMIT 100;

-- 4. 聚合查询
EXPLAIN (ANALYZE, BUFFERS, TIMING)
SELECT 
    DATE(created_at) as date,
    COUNT(*) as message_count,
    AVG(tokens) as avg_tokens,
    SUM(cost) as total_cost
FROM core.messages
WHERE created_at >= CURRENT_DATE - INTERVAL '7 days'
GROUP BY DATE(created_at);

-- 5. JOIN查询
EXPLAIN (ANALYZE, BUFFERS, TIMING)
SELECT 
    u.username,
    u.email,
    COUNT(c.id) as conversation_count,
    COUNT(m.id) as message_count,
    SUM(m.tokens) as total_tokens
FROM core.users u
LEFT JOIN core.conversations c ON u.id = c.user_id
LEFT JOIN core.messages m ON c.id = m.conversation_id
WHERE u.created_at >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY u.id, u.username, u.email
ORDER BY message_count DESC
LIMIT 100;
```

```sql
-- 创建内存监控视图
CREATE OR REPLACE VIEW admin.memory_usage AS
SELECT 
    (SELECT count(*) FROM pg_stat_activity WHERE datname = 'yyc3_test') as connections,
    (SELECT sum(shared_blks_hit) * 8192.0 / 1024 / 1024 as shared_buffer_hit_mb FROM pg_stat_database WHERE datname = 'yyc3_test'),
    (SELECT sum(blks_hit) * 8192.0 / 1024 / 1024 as local_buffer_hit_mb FROM pg_stat_database WHERE datname = 'yyc3_test'),
    (SELECT sum(temp_bytes) / 1024 / 1024 as temp_bytes_mb FROM pg_stat_database WHERE datname = 'yyc3_test'),
    now() as check_time;

-- 定期记录内存使用
CREATE TABLE admin.memory_history (
    id SERIAL PRIMARY KEY,
    connections INTEGER,
    shared_buffer_hit_mb NUMERIC,
    local_buffer_hit_mb NUMERIC,
    temp_bytes_mb NUMERIC,
    check_time TIMESTAMPTZ NOT NULL
);

-- 每分钟记录一次
CREATE OR REPLACE FUNCTION admin.record_memory_usage()
RETURNS void AS $$
BEGIN
    INSERT INTO admin.memory_history
    SELECT * FROM admin.memory_usage;
END;
$$ LANGUAGE plpgsql;

-- 设置定时任务
SELECT cron.schedule(
    'record-memory-usage',
    '* * * * *',
    $$SELECT admin.record_memory_usage()$$
);
```

### 4.4 自定义测试脚本

```python
#!/usr/bin/env python3
import psycopg2
import time
import random
from concurrent.futures import ThreadPoolExecutor

class DatabaseBenchmark:
    def __init__(self, connection_string):
        self.conn = psycopg2.connect(connection_string)
        
    def benchmark_query(self, query, iterations=1000):
        times = []
        for _ in range(iterations):
            start = time.time()
            self.conn.execute(query)
            end = time.time()
            times.append(end - start)
        
        return {
            'min': min(times),
            'max': max(times),
            'avg': sum(times) / len(times),
            'p50': sorted(times)[len(times)//2],
            'p95': sorted(times)[int(len(times)*0.95)],
            'p99': sorted(times)[int(len(times)*0.99)]
        }
    
    def benchmark_concurrent(self, query, threads=50, iterations=100):
        def run_benchmark():
            times = []
            for _ in range(iterations):
                start = time.time()
                self.conn.execute(query)
                end = time.time()
                times.append(end - start)
            return times
        
        with ThreadPoolExecutor(max_workers=threads) as executor:
            results = list(executor.map(run_benchmark, range(threads)))
        
        all_times = [t for times in results for t in times]
        return {
            'min': min(all_times),
            'max': max(all_times),
            'avg': sum(all_times) / len(all_times),
            'p50': sorted(all_times)[len(all_times)//2],
            'p95': sorted(all_times)[int(len(all_times)*0.95)],
            'p99': sorted(all_times)[int(len(all_times)*0.99)],
            'throughput': len(all_times) / sum(all_times)
        }

if __name__ == '__main__':
    conn_str = "host=localhost port=5433 dbname=yyc3_test user=yyc3_admin"
    benchmark = DatabaseBenchmark(conn_str)
    
    # 测试单查询
    query = "SELECT * FROM core.users WHERE id = gen_random_uuid()"
    result = benchmark.benchmark_query(query)
    print(f"Single query benchmark: {result}")
    
    # 测试并发查询
    result = benchmark.benchmark_concurrent(query, threads=50, iterations=100)
    print(f"Concurrent query benchmark: {result}")
```

### 5.2 性能指标收集

```sql
-- 创建性能指标收集表
CREATE TABLE admin.performance_metrics (
    id SERIAL PRIMARY KEY,
    metric_name TEXT NOT NULL,
    metric_value NUMERIC NOT NULL,
    metric_unit TEXT,
    collected_at TIMESTAMPTZ NOT NULL
);

-- 创建指标收集函数
CREATE OR REPLACE FUNCTION admin.collect_performance_metrics()
RETURNS void AS $$
BEGIN
    -- 收集连接数
    INSERT INTO admin.performance_metrics (metric_name, metric_value, metric_unit, collected_at)
    SELECT 
        'connection_count',
        count(*),
        'count',
        now()
    FROM pg_stat_activity
    WHERE datname = 'yyc3_test';
    
    -- 收集缓存命中率
    INSERT INTO admin.performance_metrics (metric_name, metric_value, metric_unit, collected_at)
    SELECT 
        'cache_hit_ratio',
        round(blks_hit::numeric / NULLIF(blks_hit + blks_read, 0) * 100, 2),
        'percent',
        now()
    FROM pg_stat_database
    WHERE datname = 'yyc3_test';
    
    -- 收集数据库大小
    INSERT INTO admin.performance_metrics (metric_name, metric_value, metric_unit, collected_at)
    SELECT 
        'database_size',
        pg_database_size('yyc3_test'),
        'bytes',
        now();
    
    -- 收集活跃查询数
    INSERT INTO admin.performance_metrics (metric_name, metric_value, metric_unit, collected_at)
    SELECT 
        'active_queries',
        count(*),
        'count',
        now()
    FROM pg_stat_activity
    WHERE datname = 'yyc3_test' AND state = 'active';
END;
$$ LANGUAGE plpgsql;

-- 设置定时收集任务
SELECT cron.schedule(
    'collect-performance-metrics',
    '* * * * *',
    $$SELECT admin.collect_performance_metrics()$$
);
```

## 6. 性能优化建议

### 6.1 查询优化

#### 6.1.1 索引优化

```sql
-- 查看未使用的索引
SELECT 
    schemaname,
    tablename,
    indexname,
    idx_scan as index_scans,
    idx_tup_read as tuples_read,
    idx_tup_fetch as tuples_fetched
FROM pg_stat_user_indexes
WHERE idx_scan = 0
AND indexrelname NOT LIKE '%_pkey'
ORDER BY pg_relation_size(indexrelid) DESC;

-- 查看低效索引
SELECT 
    schemaname,
    tablename,
    indexname,
    idx_scan as index_scans,
    pg_relation_size(indexrelid) as index_size
FROM pg_stat_user_indexes
WHERE idx_scan < 100
AND pg_relation_size(indexrelid) > 1024 * 1024  -- 大于1MB
ORDER BY index_size DESC;
```

#### 6.1.2 查询重写

```sql
-- 优化前：使用OR条件
SELECT * FROM core.messages
WHERE role = 'user' OR role = 'assistant';

-- 优化后：使用IN
SELECT * FROM core.messages
WHERE role IN ('user', 'assistant');

-- 优化前：子查询
SELECT * FROM core.users
WHERE id IN (SELECT user_id FROM core.conversations WHERE status = 'active');

-- 优化后：JOIN
SELECT DISTINCT u.*
FROM core.users u
INNER JOIN core.conversations c ON u.id = c.user_id
WHERE c.status = 'active';
```

### 8.2 性能基线管理

```sql
-- 创建性能基线表
CREATE TABLE admin.performance_baseline (
    id SERIAL PRIMARY KEY,
    test_name TEXT NOT NULL,
    metric_name TEXT NOT NULL,
    baseline_value NUMERIC NOT NULL,
    baseline_unit TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    created_by TEXT NOT NULL
);

-- 插入性能基线
INSERT INTO admin.performance_baseline (test_name, metric_name, baseline_value, baseline_unit, created_by)
VALUES 
    ('single_row_select', 'p50_latency', 0.8, 'ms', 'performance_team'),
    ('single_row_select', 'p95_latency', 1.2, 'ms', 'performance_team'),
    ('concurrent_read', 'qps', 16000, 'queries/s', 'performance_team'),
    ('concurrent_write', 'tps', 6000, 'transactions/s', 'performance_team');

-- 对比当前性能与基线
SELECT 
    pb.test_name,
    pb.metric_name,
    pb.baseline_value,
    pb.baseline_unit,
    pm.metric_value as current_value,
    pm.metric_unit as current_unit,
    ((pm.metric_value - pb.baseline_value) / pb.baseline_value * 100)::numeric(10,2) as performance_change_percent
FROM admin.performance_baseline pb
LEFT JOIN (
    SELECT 
        metric_name,
        AVG(metric_value) as metric_value,
        metric_unit
    FROM admin.performance_metrics
    WHERE collected_at >= NOW() - INTERVAL '7 days'
    GROUP BY metric_name, metric_unit
) pm ON pb.metric_name = pm.metric_name;
```
